  ROLE {
    bigint id_role PK
    varchar nama_role
  }
  USERS {
    bigint id_users PK
    bigint id_role FK
    varchar nama_lengkap
    varchar username UK
    varchar password
  }
  MEJA {
    bigint id_meja PK
    varchar no_meja
    varchar qr_code
  }
  KATEGORI_MENU {
    bigint id_kategori PK
    varchar nama_kategori
  }
  MENU {
    bigint id_menu PK
    varchar nama_menu
    text deskripsi
    int harga
    varchar gambar_menu
    enum status_ketersediaan "Tersedia|Tidak Tersedia"
    enum jenis_menu "Makanan|Minuman"
    enum kategori_makanan "Pedas|Tidak Pedas"
    enum tipe_minuman "Panas|Dingin|Keduanya"
    text komposisi
    int stock
  }
  MENU_KATEGORI {
    bigint id_menu PK, FK
    bigint id_kategori PK, FK
  }
  PESANAN {
    varchar no_pesanan PK "PS-YmdHis-XXXX"
    varchar tracking_code UK "KV-XXXXX"
    bigint id_user FK "nullable"
    bigint id_meja FK
    varchar nama_konsumen
    datetime tgl_pesanan
    int total_harga "Rupiah integer"
    enum status_pembayaran "menunggu|lunas|gagal"
    enum status_pesanan "menunggu konfirmasi|diproses|selesai"
    text catatan_pesanan
    varchar midtrans_transaction_id
    varchar qr_url
    datetime tgl_pembayaran
  }
  DETAIL_PESANAN {
    bigint id_detail PK
    varchar no_pesanan FK
    bigint id_menu FK
    int jumlah
    varchar catatan
    int subtotal
  }
  SESSIONS {
    varchar id PK
    bigint user_id "nullable, tanpa FK constraint"
    varchar ip_address
    longtext payload
    int last_activity
  }
